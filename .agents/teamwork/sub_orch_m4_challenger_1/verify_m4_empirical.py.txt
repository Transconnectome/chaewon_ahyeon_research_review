def audit_table_syntax_and_delimiters():
    text = ROADMAP_PATH.read_text(encoding="utf-8")
    lines = text.splitlines()
    in_table = False
    table_headers = []
    table_delims = []
    malformed_tables = 0
    
    for i, line in enumerate(lines):
        stripped = line.strip()
        if stripped.startswith("|") and stripped.endswith("|"):
            cols = [c.strip() for c in stripped.split("|")[1:-1]]
            if not in_table:
                in_table = True
                table_headers = cols
            elif in_table and len(table_delims) == 0:
                is_delim = all(re.match(r"^:?-+:?$", c) for c in cols)
                if is_delim:
                    table_delims = cols
                    if len(table_delims) != len(table_headers):
                        malformed_tables += 1
                else:
                    malformed_tables += 1
            else:
                if len(cols) != len(table_headers):
                    malformed_tables += 1
        else:
            in_table = False
            table_headers = []
            table_delims = []
            
    record("table_pipe_syntax", malformed_tables == 0, f"Malformed markdown table rows: {malformed_tables}", "MEDIUM")

def audit_compute_budget_arithmetic():

    # Parse Table 5.2
    exp_matches = re.findall(r'\|\s*\*\*EXP-T\d+-\d+\*\*\s*\|.*?\|\s*\*\*([0-9,]+)\s*GPUh\*\*\s*\|', text)
    individual_gpuh = [int(m.replace(',', '')) for m in exp_matches]
    actual_sum = sum(individual_gpuh)
    
    tot_match = re.search(r'\|\s*\*\*합계\*\*\s*\|.*?\|\s*\*\*([0-9,]+)\s*GPUh\*\*\s*\|', text)
    if tot_match:
        reported_sum = int(tot_match.group(1).replace(',', ''))
    else:
        reported_sum = None
        
    passed = (reported_sum == actual_sum)
    msg = f"Reported total: {reported_sum} GPUh, Actual sum of 20 items: {actual_sum} GPUh (Delta: {actual_sum - (reported_sum or 0)} GPUh)"
    record("compute_budget_table_sum", passed, msg, "HIGH")
    
    # Check Aurora allocation
    aurora_matches = re.findall(r'\|\s*\*\*EXP-T\d+-\d+\*\*\s*\|.*?ALCF Aurora.*?\|\s*\*\*([0-9,]+)\s*GPUh\*\*\s*\|', text)
    aurora_sum = sum(int(m.replace(',', '')) for m in aurora_matches)

def audit_statistical_fdr_and_claims():
    text = ROADMAP_PATH.read_text(encoding="utf-8")
    # Extract RAW_DELTAS from text if present
    match = re.search(r'RAW_DELTAS\s*=\s*np\.array\(\[\s*([\s\S]*?)\]\)', text)
    if not match:
        record("raw_deltas_presence", False, "Could not extract RAW_DELTAS matrix from text", "HIGH")
        return
    
    matrix_str = match.group(1).strip()
    rows = []
    for line in matrix_str.splitlines():
        line = line.strip().rstrip(',')
        if line.startswith('[') and line.endswith(']'):
            vals = [float(x.strip()) for x in line[1:-1].split(',')]
            rows.append(vals)
            
    deltas = np.array(rows)
    df = deltas.shape[1] - 1
    means = np.mean(deltas, axis=1)
    sems = stats.sem(deltas, axis=1)
    margin = -0.005
    
    t_stats = (means - margin) / sems
    p_noninf = 1.0 - stats.t.cdf(t_stats, df)
    
    n_tasks = len(p_noninf)
    sorted_idx = np.argsort(p_noninf)
    sorted_p = p_noninf[sorted_idx]
    q_vals = np.zeros(n_tasks)
    q_vals[sorted_idx] = np.minimum.accumulate((sorted_p * n_tasks / np.arange(1, n_tasks + 1))[::-1])[::-1]
    q_vals = np.clip(q_vals, 0, 1)
    
    passing_fdr = np.sum(q_vals < 0.05)
    passing_bonf = np.sum(p_noninf < (0.05 / n_tasks))
    
    record("fdr_calculation_verified", True, f"FDR evaluated on {n_tasks} tasks: {passing_fdr}/{n_tasks} pass FDR, {passing_bonf}/{n_tasks} pass Bonferroni")
    
    # Check claim: "across all individual subtasks"
    all_subtasks_claimed = "across all individual subtasks" in text
    if all_subtasks_claimed and passing_fdr < n_tasks:
        record("abstract_claim_soundness", False, f"Abstract claims non-inferiority 'across all individual subtasks', but empirical math shows only {passing_fdr}/{n_tasks} pass (Tasks 9, 10, 13, 14 fail)", "HIGH")
    else:
        record("abstract_claim_soundness", True, "Claims accurately reflect FDR pass rate", "LOW")
